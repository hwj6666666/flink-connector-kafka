% 行内标识符：版式上等价于 Markdown 的 `code`（等宽），但学位论文通常不用彩色底纹。
% 断行策略：不在字母中间随意断开；仅在分隔符（. _ - = 等）之后以及小写→大写（CamelCase 边界）处允许换行，
% 避免出现 “StreamPa…tternSubscriber” 这类难看断字。
\ExplSyntaxOn
\bool_new:N \l__sjtu_id_first_bool
\tl_new:N \l__sjtu_id_prev_tl
\bool_new:N \l__sjtu_id_delim_match_bool
\clist_const:Nn \c__sjtu_id_delim_clist
  { { . } { _ } { - } { = } { : } { , } { ; } { ( } { ) } { < } { > } { [ } { ] } { / } { | } }
\prg_new_conditional:Npnn \__sjtu_id_ascii_lower:n #1 { p }
  {
    \int_compare:nNnTF { `#1 } < { 97 } { \prg_return_false: }
      {
        \int_compare:nNnTF { `#1 } > { 122 } { \prg_return_false: }
          { \prg_return_true: }
      }
  }
\prg_new_conditional:Npnn \__sjtu_id_ascii_upper:n #1 { p }
  {
    \int_compare:nNnTF { `#1 } < { 65 } { \prg_return_false: }
      {
        \int_compare:nNnTF { `#1 } > { 90 } { \prg_return_false: }
          { \prg_return_true: }
      }
  }
\cs_new_protected:Npn \__sjtu_id_delimiter_if:nTF #1#2#3
  {
    \bool_set_false:N \l__sjtu_id_delim_match_bool
    \clist_map_inline:Nn \c__sjtu_id_delim_clist
      {
        \str_if_eq:nnT {#1} {##1}
          {
            \bool_set_true:N \l__sjtu_id_delim_match_bool
            \clist_map_break:n { }
          }
      }
    \bool_if:NTF \l__sjtu_id_delim_match_bool {#2} {#3}
  }
\cs_new_protected:Npn \__sjtu_id_maybe_break:nn #1#2
  {
    \__sjtu_id_delimiter_if:nTF {#1}
      { \allowbreak }
      {
        \bool_lazy_and:nnTF
          { \__sjtu_id_ascii_lower_p:n {#1} }
          { \__sjtu_id_ascii_upper_p:n {#2} }
          { \allowbreak }
          { }
      }
  }
\cs_new_protected:Npn \__sjtu_id_format:n #1
  {
    \bool_set_true:N \l__sjtu_id_first_bool
    \tl_clear:N \l__sjtu_id_prev_tl
    \str_map_inline:nn {#1}
      {
        \bool_if:NTF \l__sjtu_id_first_bool
          {
            ##1
            \bool_set_false:N \l__sjtu_id_first_bool
            \tl_set:Nn \l__sjtu_id_prev_tl {##1}
          }
          {
            \exp_args:No \__sjtu_id_maybe_break:nn \l__sjtu_id_prev_tl {##1}
            ##1
            \tl_set:Nn \l__sjtu_id_prev_tl {##1}
          }
      }
  }
\cs_new_protected:Npn \sjtu_thesis_id:n #1
  {
    \texttt { \exp_args:Ne \__sjtu_id_format:n { \tl_to_str:n {#1} } }
  }
\NewDocumentCommand \id { m } { \sjtu_thesis_id:n {#1} }
